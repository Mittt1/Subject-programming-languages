open System

// Функция для преобразования символа в цифру или оставляем символ как есть
// - contrary: bool - флаг, указывающий, нужно ли применить знак минуса к цифре
let digit_conversion (c: char) (contrary: bool) =
    if c >= '0' && c <= '9' then
        // Если символ является цифрой, преобразуем его в число
        let a = int c - int '0'  // a - цифра 
        // Если флаг contrary установлен, применяем знак минуса к цифре
        if contrary then -a else a
    else
        // Если символ не является цифрой, выбрасываем исключение
        failwithf "Недопустимый символ: %c" c

// Функция для преобразования строки в список цифр и символов (с учетом знака)
// Параметры:
// - s: string - входная строка
let stringToDigitList (s: string) =
    // Проверяем, начинается ли строка с минуса
    let contrary = s.StartsWith("-")
    // Если строка начинается с минуса, удаляем его; иначе оставляем строку как есть
    let n = if contrary then s.Substring(1) else s // n - номер строки
    // Получаем первый символ строки после удаления минуса (если он был)
    let a0 =   // a0 - первая цифра
        if n.Length > 0 then 
            // Если строка не пустая, преобразуем первый символ в цифру
            Some (digit_conversion n.[0] contrary)
        else 
            // Если строка пустая после удаления минуса, возвращаем None
            None // Обработка случая пустой строки после удаления минуса
    
    // Обрабатываем результат преобразования первого символа
    match a0 with
    | Some first ->
        // Если первый символ успешно преобразован, создаем список
        [
            // Добавляем первый элемент в список
            yield first
            // Обрабатываем остальные символы строки
            for i in 1 .. n.Length - 1 do
                let c = n.[i]
                // Если символ является цифрой, преобразуем его в чифру
                if c >= '0' && c <= '9' then
                    yield digit_conversion c false
                else
                    // Если символ не является цифрой, добавляем его как есть (в виде ASCII-кода)
                    yield (int c) // Добавляем символ как есть
        ]
    | None -> [] // Возвращаем пустой список, если строка была пустой или содержала только минус

printfn "8 вариант: Сформировать список из цифр числа."

// Ввод числа с клавиатуры
printfn "Введите число: "
let input = Console.ReadLine()

// Преобразование введённого числа в список цифр и символов
let List_digit  = stringToDigitList (input.Trim()) // .Trim - для удаления (завершающих) пробелов

// Вывод списка цифр и символов
printfn "Список цифр и символов: %A" List_digit
