open System

// Функция для добавления элемента в начало списка 
let addElement element list = element :: list

// Функция для удаления элемента из списка 
let rec remove element list =
    match list with
    | [] -> []
    | head :: tail when head = element -> tail
    | head :: tail -> head :: (remove element tail)

// Функция для поиска элемента в списке 
let rec find element list =
    match list with
    | [] -> false
    | head :: tail when head = element -> true
    | head :: tail -> find element tail

// Функция для сцепления двух списков 
let rec concat list1 list2 =
    match list1 with
    | [] -> list2
    | head :: tail -> head :: (concat tail list2)

// Функция для получения элемента по индексу 
let rec getElementByIndex index list =
    let rec getElement index list acc =
        match list with
        | [] -> None
        | head :: tail when acc = index -> Some head
        | head :: tail -> getElement index tail (acc + 1)
    getElement index list 0
